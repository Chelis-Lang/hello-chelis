% Definite integral of a free variable f(x) over [a, b]:
\begin{equation}\label{eq:integral}
\int_a^b x \, dx
\end{equation}

% Gradient of an objective L w.r.t. parameters \theta:
\begin{equation}\label{eq:grad}
\nabla L
\end{equation}

% Partial derivative:
\begin{equation}\label{eq:partial}
\frac{\partial y}{\partial x}
\end{equation}

% Inner product as a sum:
\begin{equation}\label{eq:innerprod}
\sum_i w_i x_i
\end{equation}
