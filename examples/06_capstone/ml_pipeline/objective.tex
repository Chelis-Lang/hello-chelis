% Logistic regression negative log-likelihood. Sigmoid is implicit in the
% \sigma() macro that Octant expands to Std.Nn.Activation.sigmoid.
\begin{equation}\label{eq:nll}
L(w, b) = - \frac{1}{n} \sum_i \left[ y_i \log \sigma(w \cdot x_i + b) + (1 - y_i) \log (1 - \sigma(w \cdot x_i + b)) \right]
\end{equation}
